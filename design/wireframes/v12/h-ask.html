Query it ran42 ms
SELECT day, SUM(pageviews) AS pageviews, SUM(sessions) AS sessions
FROM ask_daily
WHERE day >= now() - interval '7 days'
GROUP BY day ORDER BY pageviews DESC LIMIT 3;
